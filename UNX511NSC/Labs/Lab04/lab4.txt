#include <iostream>
#include <fcntl.h>
#include <unistd.h>
#include <sys/ioctl.h>
#include <linux/fb.h>
#include <errno.h>
#include <string.h>
#include <fstream>

using namespace std;

void redirectError() {
    int log_fd = open("Screen.log", O_WRONLY | O_CREAT | O_TRUNC, 0644);
    if (log_fd < 0) {
        cerr << "Failed to open Screen.log: " << strerror(errno) << endl;
        exit(1);
    }
    dup2(log_fd, STDERR_FILENO); // Redirect cerr to Screen.log
    close(log_fd);
}

int main() {
    redirectError();

    int fb_fd = open("/dev/fb0", O_RDONLY | O_NONBLOCK);
    if (fb_fd < 0) {
        cerr << "Error opening /dev/fb0: " << strerror(errno) << endl;
        return 1;
    }

    int choice;
    do {
        cout << "\nSelect an option:\n"
            << "1. Fixed Screen Info\n"
            << "2. Variable Screen Info\n"
            << "0. Exit\n"
            << "Choice: ";
        cin >> choice;

        if (choice == 1) {
            struct fb_fix_screeninfo fix_info;
            if (ioctl(fb_fd, FBIOGET_FSCREENINFO, &fix_info) < 0) {
                cerr << "Error getting fixed screen info: " << strerror(errno) << endl;
                continue;
            }

            cout << "\nFixed Screen Info:\n";
            cout << "Visual Type: " << fix_info.visual << endl;        // FB_VISUAL_*
            cout << "Accelerator: " << fix_info.accel << endl;        // FB_ACCEL_*
            cout << "Capabilities: " << fix_info.capabilities << endl; // FB_CAP_*

            // Add these values as comments in your code submission
        }
        else if (choice == 2) {
            struct fb_var_screeninfo var_info;
            if (ioctl(fb_fd, FBIOGET_VSCREENINFO, &var_info) < 0) {
                cerr << "Error getting variable screen info: " << strerror(errno) << endl;
                continue;
            }

            cout << "\nVariable Screen Info:\n";
            cout << "X Resolution: " << var_info.xres << endl;
            cout << "Y Resolution: " << var_info.yres << endl;
            cout << "Bits Per Pixel: " << var_info.bits_per_pixel << endl;

            // Add these values as comments in your code submission
        }

    } while (choice != 0);

    close(fb_fd);
    return 0;
}
